#include <stdio.h>
#include <math.h>

void main() {
    float radius, length, width, height, area, perimeter, base, slant_height;
    char first_choice, second_choice;

    printf("---Geometry Calculator---\n");
    printf("C: Circle\nR: Rectangle\nT: Triangle\n");
    printf("Select a shape: ");
    scanf(" %c", &first_choice);

    printf("A: Area\nP: Perimeter\n");
    printf("Choose an operation: ");
    scanf(" %c", &second_choice);

    switch (first_choice)
    {
    case 'C':
        printf("Enter radius of the circle: ");
        scanf(" %f", &radius);
        switch (second_choice)
        {
        case 'A':
            area = 3.14 * (radius * radius);
            printf("Area of the circle is %.2f", area);
            break;
        
        case 'P':
            perimeter = 2 * 3.14 * radius;
            printf("Perimeter of the circle is %.2f", perimeter);
            break;
        
        default:
            printf("Invalid operation, kindly enter A or P");
            break;
        }
        break;
    case 'R':
        printf("Enter length of the rectangle: ");
        scanf(" %f", &length);

        printf("Enter width of the rectangle: ");
        scanf(" %f", &width);

        switch (second_choice)
        {
        case 'A':
            area = length * width;
            printf("Area of the rectangle is %.2f", area);
            break;
        case 'P':
            perimeter = (2 * length) + (2 * width);
            printf("Perimeter of the rectangle is %.2f", perimeter);
            break;
        
        default:
            printf("Invalid operation, kindly enter A or P");
            break;
        }
        break;
    
    case 'T':
        printf("Enter the base of the triangle: ");
        scanf(" %f", &base);

        printf("Enter the height of the triangle: ");
        scanf(" %f", &height);

        switch (second_choice)
        {
        case 'A':
            area = 0.5 * base * height;
            printf("Area of the triangle is %.2f", area);
            break;
        case 'P':
            perimeter = base + height + sqrt((base * base) + (height * height));
            printf("Perimeter of the triangle is %.2f", perimeter);
            break;
        
        default:
            printf("Invalid operation, kindly enter A or P");
            break;
        }
        break;
    default:
        printf("Invalid shape, kindle enter C, R or T");
        break;
    }
}
